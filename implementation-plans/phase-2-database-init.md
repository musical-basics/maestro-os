# Phase 2: Database Initialization & Typings

> **Goal:** Build the offline SQLite database inside the Electron Main process, create all tables with proper relationships, seed the 15 pipeline stages, and establish strict full-stack type safety.
>
> **Depends on:** Phase 1 completed (Electron scaffold exists, `better-sqlite3` installed).

## Step 1 — Define Shared TypeScript Interfaces

### What to do

Create `src/shared/types.ts` (this file is imported by BOTH renderer AND main process code):

```typescript
// src/shared/types.ts

export interface Snippet {
  id: string;                // UUID (generated by Node.crypto.randomUUID())
  title: string;             // e.g., "Sad Piano Arpeggio"
  filePath: string;          // Absolute local path: /Users/name/Music/Memos/memo.m4a
  assetType: 'audio' | 'midi' | 'sheet' | 'video';
  keySignature: string | null;
  bpm: number | null;
  createdAt: string;         // ISO 8601 timestamp
}

export interface Project {
  id: string;                // UUID
  title: string;             // e.g., "Neon Sonata"
  currentStageId: number;    // 1 through 15
  masterDirectory: string;   // Absolute path: /Users/name/Music/MaestroOS/Neon_Sonata/
  lastTouchedAt: string;     // ISO timestamp (updated by chokidar)
  createdAt: string;         // ISO timestamp
}

export interface ProjectAsset {
  id: string;                // UUID
  projectId: string;         // FK → projects.id
  stageId: number;           // FK → pipeline_stages.id
  filePath: string;          // Absolute local path
  assetType: string;         // File extension, e.g., '.logicx', '.wav', '.mp4'
  lastModified: string;      // ISO timestamp
}

export interface PipelineStage {
  id: number;                // 1 through 15
  stageName: string;         // e.g., "Audio Editing"
  gateRequirement: string | null;  // e.g., '.wav', or null for conceptual stages
}

export interface ProjectSnippet {
  projectId: string;         // FK → projects.id
  snippetId: string;         // FK → snippets.id
}
```

### Why
This is the single source of truth for all data shapes. The same file is imported by:
- `electron/main.ts` (IPC handler return types)
- `electron/preload.ts` (bridge types)
- `src/types/electron.d.ts` (renderer-side `window.api` types)
- All React components and zustand stores

## Step 2 — Initialize SQLite Connection

### What to do

Create `electron/database/connection.ts`:

```typescript
import Database from 'better-sqlite3';
import path from 'path';
import { app } from 'electron';

let db: Database.Database | null = null;

export function getDatabase(): Database.Database {
  if (db) return db;

  // Store the .db file in the macOS app data directory:
  // ~/Library/Application Support/MaestroOS/maestro.db
  const dbPath = path.join(app.getPath('userData'), 'maestro.db');

  db = new Database(dbPath);

  // Enable WAL mode for better concurrent read performance
  db.pragma('journal_mode = WAL');

  // Enable foreign key enforcement (SQLite has this OFF by default!)
  db.pragma('foreign_keys = ON');

  return db;
}

export function closeDatabase(): void {
  if (db) {
    db.close();
    db = null;
  }
}
```

### Critical details
- **`app.getPath('userData')`** resolves to `~/Library/Application Support/MaestroOS/` on macOS. This is the correct protected location for app data — NOT the user's project folders.
- **WAL mode** is critical for performance — without it, reads block on writes.
- **Foreign keys are OFF by default** in SQLite. You **must** call `db.pragma('foreign_keys = ON')` or the FK constraints in your schema are silently ignored.

## Step 3 — Create Core Tables

### What to do

Create `electron/database/schema.ts`:

```typescript
import { getDatabase } from './connection';

export function initializeSchema(): void {
  const db = getDatabase();

  // Run all table creation in a single transaction for atomicity
  db.exec(`
    -----------------------------------------------------------
    -- 1. snippets: Holding pen for unassigned creative ideas
    -----------------------------------------------------------
    CREATE TABLE IF NOT EXISTS snippets (
      id            TEXT PRIMARY KEY,
      title         TEXT NOT NULL,
      file_path     TEXT NOT NULL UNIQUE,
      asset_type    TEXT NOT NULL CHECK(asset_type IN ('audio', 'midi', 'sheet', 'video')),
      key_signature TEXT,
      bpm           INTEGER,
      created_at    DATETIME DEFAULT CURRENT_TIMESTAMP
    );

    -----------------------------------------------------------
    -- 2. projects: Master tracker for active compositions
    -----------------------------------------------------------
    CREATE TABLE IF NOT EXISTS projects (
      id               TEXT PRIMARY KEY,
      title            TEXT NOT NULL,
      current_stage_id INTEGER NOT NULL DEFAULT 1,
      master_directory TEXT NOT NULL,
      last_touched_at  DATETIME DEFAULT CURRENT_TIMESTAMP,
      created_at       DATETIME DEFAULT CURRENT_TIMESTAMP,
      FOREIGN KEY (current_stage_id) REFERENCES pipeline_stages(id)
    );

    -----------------------------------------------------------
    -- 3. pipeline_stages: The 15 rigid stages (lookup/enum)
    -----------------------------------------------------------
    CREATE TABLE IF NOT EXISTS pipeline_stages (
      id               INTEGER PRIMARY KEY,
      stage_name       TEXT NOT NULL,
      gate_requirement TEXT  -- NULL means conceptual stage (no file needed)
    );

    -----------------------------------------------------------
    -- 4. project_snippets: Many-to-Many junction
    -----------------------------------------------------------
    CREATE TABLE IF NOT EXISTS project_snippets (
      project_id TEXT NOT NULL,
      snippet_id TEXT NOT NULL,
      PRIMARY KEY (project_id, snippet_id),
      FOREIGN KEY (project_id) REFERENCES projects(id) ON DELETE CASCADE,
      FOREIGN KEY (snippet_id) REFERENCES snippets(id) ON DELETE CASCADE
    );

    -----------------------------------------------------------
    -- 5. project_assets: Files linked to satisfy Asset Gates
    -----------------------------------------------------------
    CREATE TABLE IF NOT EXISTS project_assets (
      id            TEXT PRIMARY KEY,
      project_id    TEXT NOT NULL,
      stage_id      INTEGER NOT NULL,
      file_path     TEXT NOT NULL,
      asset_type    TEXT NOT NULL,
      last_modified DATETIME NOT NULL,
      FOREIGN KEY (project_id) REFERENCES projects(id) ON DELETE CASCADE,
      FOREIGN KEY (stage_id) REFERENCES pipeline_stages(id)
    );
  `);
}
```

### Critical details
- **`CREATE TABLE IF NOT EXISTS`** — the app calls this on every launch. It's a no-op if tables exist.
- **`ON DELETE CASCADE`** on junction tables — if you delete a project, its linked snippets and assets are automatically cleaned up.
- **`CHECK` constraint on `asset_type`** in `snippets` — enforces only valid types.
- The `pipeline_stages` table is created BEFORE `projects` because `projects.current_stage_id` has a FK to it.

## Step 4 — Seed the 15 Pipeline Stages

### What to do

Create `electron/database/seed.ts`:

```typescript
import { getDatabase } from './connection';

export function seedPipelineStages(): void {
  const db = getDatabase();

  // Only seed if the table is empty (prevents duplicate rows on re-launch)
  const count = db.prepare('SELECT COUNT(*) as n FROM pipeline_stages').get() as { n: number };
  if (count.n > 0) return;

  const insert = db.prepare(
    'INSERT INTO pipeline_stages (id, stage_name, gate_requirement) VALUES (?, ?, ?)'
  );

  const stages: [number, string, string | null][] = [
    [1,  'Idea (Vault)',                    null],       // Conceptual — no file gate
    [2,  'Half-Finished Composition',       null],       // Conceptual
    [3,  'Completed Composition',           '.mid'],     // MIDI or sheet music file
    [4,  'Practicing',                      null],       // Conceptual — manual "Mark Complete"
    [5,  'Recording',                       '.wav'],     // Raw audio recording
    [6,  'Audio Editing',                   '.logicx'],  // Logic Pro project file
    [7,  'Mixing',                          '.wav'],     // Bounced mix
    [8,  'Mastering',                       '.wav'],     // Final mastered WAV
    [9,  'Video Recording',                 '.mp4'],     // Raw video
    [10, 'Video Editing',                   '.mp4'],     // Edited video
    [11, 'Thumbnail Creation',             '.png'],     // Thumbnail image
    [12, 'Description & Metadata Writing',  null],       // Conceptual — manual "Mark Complete"
    [13, 'Social Media Assets',             '.mp4'],     // Shorts/Reels cutdowns
    [14, 'Scheduled/Uploading',             null],       // Conceptual — manual "Mark Complete"
    [15, 'Released',                        null],       // Terminal state
  ];

  const insertMany = db.transaction(() => {
    for (const [id, name, gate] of stages) {
      insert.run(id, name, gate);
    }
  });

  insertMany();
}
```

### Critical details
- The 15 stages come from **`docs/prd.md` → 2.3 The Pipeline Stages**.
- The `gate_requirement` values come from **`docs/data-structure.md` → Table: pipeline_stages**.
- Stages with `null` gate requirements (1, 2, 4, 12, 14, 15) are "conceptual" — the user progresses past them with a manual "Mark as Complete" button (built in Phase 6).
- The seed runs inside a **transaction** for atomicity — if any insert fails, they all roll back.
- The `COUNT(*)` guard prevents duplicate seeding if the user opens the app multiple times.

## Step 5 — Wire Database Initialization into App Startup

### What to do

In `electron/main.ts`, call the database initialization **before** creating the BrowserWindow:

```typescript
import { initializeSchema } from './database/schema';
import { seedPipelineStages } from './database/seed';
import { closeDatabase } from './database/connection';

app.whenReady().then(() => {
  // 1. Initialize database schema (creates tables if they don't exist)
  initializeSchema();

  // 2. Seed the 15 pipeline stages (no-op if already seeded)
  seedPipelineStages();

  // 3. Create the BrowserWindow (existing code from Phase 1)
  createWindow();
});

// Clean up database connection on quit
app.on('will-quit', () => {
  closeDatabase();
});
```

### Why
The database and seeds must be ready **before** any IPC handlers are called by the renderer. Closing the connection on `will-quit` prevents SQLite file corruption.

## Step 6 — Create Database Helper/DAO Layer

### What to do

Create `electron/database/dao.ts` with typed helper functions that map between SQLite rows (snake_case) and TypeScript interfaces (camelCase):

```typescript
import { getDatabase } from './connection';
import { Snippet, Project, ProjectAsset, PipelineStage } from '../../src/shared/types';
import crypto from 'crypto';

// ─── Row Mappers ────────────────────────────────────────────
// SQLite returns snake_case columns. These functions convert to camelCase interfaces.

function mapSnippetRow(row: any): Snippet {
  return {
    id: row.id,
    title: row.title,
    filePath: row.file_path,
    assetType: row.asset_type,
    keySignature: row.key_signature,
    bpm: row.bpm,
    createdAt: row.created_at,
  };
}

function mapProjectRow(row: any): Project {
  return {
    id: row.id,
    title: row.title,
    currentStageId: row.current_stage_id,
    masterDirectory: row.master_directory,
    lastTouchedAt: row.last_touched_at,
    createdAt: row.created_at,
  };
}

function mapAssetRow(row: any): ProjectAsset {
  return {
    id: row.id,
    projectId: row.project_id,
    stageId: row.stage_id,
    filePath: row.file_path,
    assetType: row.asset_type,
    lastModified: row.last_modified,
  };
}

function mapStageRow(row: any): PipelineStage {
  return {
    id: row.id,
    stageName: row.stage_name,
    gateRequirement: row.gate_requirement,
  };
}

// ─── Snippets ───────────────────────────────────────────────

export function getAllSnippets(): Snippet[] {
  const rows = getDatabase().prepare('SELECT * FROM snippets ORDER BY created_at DESC').all();
  return rows.map(mapSnippetRow);
}

export function createSnippet(data: Omit<Snippet, 'id' | 'createdAt'>): Snippet {
  const id = crypto.randomUUID();
  getDatabase().prepare(`
    INSERT INTO snippets (id, title, file_path, asset_type, key_signature, bpm)
    VALUES (?, ?, ?, ?, ?, ?)
  `).run(id, data.title, data.filePath, data.assetType, data.keySignature, data.bpm);
  return mapSnippetRow(getDatabase().prepare('SELECT * FROM snippets WHERE id = ?').get(id));
}

export function deleteSnippet(id: string): void {
  getDatabase().prepare('DELETE FROM snippets WHERE id = ?').run(id);
}

// ─── Projects ───────────────────────────────────────────────

export function getAllProjects(): Project[] {
  const rows = getDatabase().prepare('SELECT * FROM projects ORDER BY last_touched_at DESC').all();
  return rows.map(mapProjectRow);
}

export function getProjectById(id: string): Project | null {
  const row = getDatabase().prepare('SELECT * FROM projects WHERE id = ?').get(id);
  return row ? mapProjectRow(row) : null;
}

export function createProject(title: string, masterDirectory: string, snippetIds: string[]): Project {
  const db = getDatabase();
  const id = crypto.randomUUID();

  const txn = db.transaction(() => {
    // Insert the project at Stage 2 (Half-Finished Composition)
    db.prepare(`
      INSERT INTO projects (id, title, current_stage_id, master_directory)
      VALUES (?, ?, 2, ?)
    `).run(id, title, masterDirectory);

    // Link all selected snippets via the junction table
    const linkStmt = db.prepare('INSERT INTO project_snippets (project_id, snippet_id) VALUES (?, ?)');
    for (const snippetId of snippetIds) {
      linkStmt.run(id, snippetId);
    }
  });

  txn();
  return mapProjectRow(db.prepare('SELECT * FROM projects WHERE id = ?').get(id));
}

export function updateProjectStage(projectId: string, newStageId: number): void {
  getDatabase().prepare('UPDATE projects SET current_stage_id = ? WHERE id = ?').run(newStageId, projectId);
}

export function updateProjectLastTouched(projectId: string): void {
  getDatabase().prepare('UPDATE projects SET last_touched_at = CURRENT_TIMESTAMP WHERE id = ?').run(projectId);
}

// ─── Assets ─────────────────────────────────────────────────

export function getAssetsByProject(projectId: string): ProjectAsset[] {
  const rows = getDatabase().prepare('SELECT * FROM project_assets WHERE project_id = ?').all(projectId);
  return rows.map(mapAssetRow);
}

export function linkAsset(data: Omit<ProjectAsset, 'id'>): ProjectAsset {
  const id = crypto.randomUUID();
  getDatabase().prepare(`
    INSERT INTO project_assets (id, project_id, stage_id, file_path, asset_type, last_modified)
    VALUES (?, ?, ?, ?, ?, ?)
  `).run(id, data.projectId, data.stageId, data.filePath, data.assetType, data.lastModified);
  return mapAssetRow(getDatabase().prepare('SELECT * FROM project_assets WHERE id = ?').get(id));
}

// ─── Pipeline Stages ────────────────────────────────────────

export function getAllStages(): PipelineStage[] {
  const rows = getDatabase().prepare('SELECT * FROM pipeline_stages ORDER BY id ASC').all();
  return rows.map(mapStageRow);
}

export function getStageById(id: number): PipelineStage | null {
  const row = getDatabase().prepare('SELECT * FROM pipeline_stages WHERE id = ?').get(id);
  return row ? mapStageRow(row) : null;
}
```

### Why
This DAO layer is the **only** place raw SQL lives. IPC handlers in `main.ts` call these functions and return typed objects. The row mappers handle the snake_case→camelCase conversion so the renderer never sees SQL column names.
